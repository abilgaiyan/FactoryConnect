CREATE TABLE dbo.OperationalMetricConflictWitnessArchive
(
    WitnessId binary(32) NOT NULL,
    WitnessCodecVersion smallint NOT NULL,
    WitnessPayload varbinary(max) NOT NULL,
    MetricAggregationProcessorRowId bigint NOT NULL,
    MetricInputStreamRowId bigint NOT NULL,
    SourceRevisionPosition decimal(20,0) NOT NULL,
    ContributionFactRowId bigint NOT NULL,
    ContributionPosition decimal(20,0) NOT NULL,

    CONSTRAINT PK_OMCWA PRIMARY KEY (WitnessId),
    CONSTRAINT CK_OMCWA_Codec CHECK (WitnessCodecVersion = 1),
    CONSTRAINT CK_OMCWA_Length CHECK
        (DATALENGTH(WitnessPayload) BETWEEN 1 AND 65536),
    CONSTRAINT CK_OMCWA_Positions CHECK
        (SourceRevisionPosition BETWEEN 1 AND 18446744073709551615
         AND ContributionPosition BETWEEN 1 AND SourceRevisionPosition),
    CONSTRAINT FK_OMCWA_Source FOREIGN KEY
        (MetricAggregationProcessorRowId, MetricInputStreamRowId)
        REFERENCES dbo.MetricAggregationProcessor
        (MetricAggregationProcessorRowId, MetricInputStreamRowId),
    CONSTRAINT FK_OMCWA_Revision FOREIGN KEY
        (MetricAggregationProcessorRowId, SourceRevisionPosition)
        REFERENCES dbo.MetricAggregationRevision
        (MetricAggregationProcessorRowId, Position),
    CONSTRAINT FK_OMCWA_Contribution FOREIGN KEY
        (MetricAggregationProcessorRowId, ContributionFactRowId)
        REFERENCES dbo.MetricAggregationContribution
        (MetricAggregationProcessorRowId, MetricInputFactRowId),
    CONSTRAINT FK_OMCWA_FactPosition FOREIGN KEY
        (ContributionFactRowId, MetricInputStreamRowId, ContributionPosition)
        REFERENCES dbo.MetricInputFact
        (MetricInputFactRowId, MetricInputStreamRowId, Position)
);

CREATE TABLE dbo.OperationalMetricPublicationResultArchive
(
    ResultId binary(32) NOT NULL,
    OperationalMetricProjectionProcessorRowId bigint NOT NULL,
    PublicationFormatVersion smallint NOT NULL,
    MetricAggregationProcessorRowId bigint NOT NULL,
    MetricInputStreamRowId bigint NOT NULL,
    SourceRevisionPosition decimal(20,0) NOT NULL,
    EvaluationKeyHash binary(32) NOT NULL,
    EvaluationKeyBinary varbinary(max) NOT NULL,
    EvaluationPolicyId nvarchar(64) COLLATE Latin1_General_100_BIN2 NOT NULL,
    ResultCodecVersion smallint NOT NULL,
    ResultPayload varbinary(max) NOT NULL,
    Status tinyint NOT NULL,
    ReasonCode tinyint NULL,
    ConflictWitnessId binary(32) NULL,

    CONSTRAINT PK_OMPRA PRIMARY KEY (ResultId),
    CONSTRAINT UQ_OMPRA_Slot UNIQUE
        (OperationalMetricProjectionProcessorRowId,
         EvaluationKeyHash, SourceRevisionPosition),
    CONSTRAINT UQ_OMPRA_RetainedBinding UNIQUE
        (ResultId, OperationalMetricProjectionProcessorRowId,
         EvaluationKeyHash, SourceRevisionPosition),
    CONSTRAINT CK_OMPRA_Format CHECK
        (PublicationFormatVersion = 2 AND ResultCodecVersion = 2),
    CONSTRAINT CK_OMPRA_Policy CHECK
        (EvaluationPolicyId = N'availability-scope-containment/1'),
    CONSTRAINT CK_OMPRA_KeyLength CHECK
        (DATALENGTH(EvaluationKeyBinary) BETWEEN 1 AND 6992),
    CONSTRAINT CK_OMPRA_PayloadLength CHECK
        (DATALENGTH(ResultPayload) BETWEEN 1 AND 16777216),
    CONSTRAINT CK_OMPRA_Position CHECK
        (SourceRevisionPosition BETWEEN 1 AND 18446744073709551615),
    CONSTRAINT CK_OMPRA_StatusReason CHECK
        ((Status = 0 AND ReasonCode IS NULL)
         OR (Status = 1 AND ReasonCode IS NOT NULL
             AND ReasonCode IN (2,3,4,6))
         OR (Status = 2 AND ReasonCode IS NOT NULL
             AND ReasonCode IN (0,1,5))),
    CONSTRAINT CK_OMPRA_WitnessShape CHECK
        ((ReasonCode IS NOT NULL AND ReasonCode = 6 AND ConflictWitnessId IS NOT NULL)
         OR ((ReasonCode IS NULL OR ReasonCode <> 6)
             AND ConflictWitnessId IS NULL)),
    CONSTRAINT FK_OMPRA_ProcessorFormat FOREIGN KEY
        (OperationalMetricProjectionProcessorRowId, PublicationFormatVersion)
        REFERENCES dbo.OperationalMetricProjectionProcessor
        (OperationalMetricProjectionProcessorRowId, PublicationFormatVersion),
    CONSTRAINT FK_OMPRA_ProcessorSource FOREIGN KEY
        (OperationalMetricProjectionProcessorRowId,
         MetricAggregationProcessorRowId, MetricInputStreamRowId)
        REFERENCES dbo.OperationalMetricProjectionProcessor
        (OperationalMetricProjectionProcessorRowId,
         MetricAggregationProcessorRowId, MetricInputStreamRowId),
    CONSTRAINT FK_OMPRA_Revision FOREIGN KEY
        (MetricAggregationProcessorRowId, SourceRevisionPosition)
        REFERENCES dbo.MetricAggregationRevision
        (MetricAggregationProcessorRowId, Position),
    CONSTRAINT FK_OMPRA_RootWitness FOREIGN KEY (ConflictWitnessId)
        REFERENCES dbo.OperationalMetricConflictWitnessArchive (WitnessId)
);

             AND EvaluationPolicyId = N'availability-scope-containment/1'
             AND ContainmentResultId IS NOT NULL)),
